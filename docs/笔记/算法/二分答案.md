# 二分答案

## 整数二分模板

```c++
#include<bits/stdc++.h>
#define ll long long
using namespace std;
ll f(int x){
    ll sum=0;
    for(...){
        if(...){
            sum+=...;
        }
    }
    return sum;
}
int main(){
    int l,r,ans;
    while(l<=r){
        int mid=(l+r)/2;
        if(f(mid)>=x){
            l=mid+1;
        }
        else{
            r=mid-1;
        }
    }
    cout<<ans<<'\n';
    return 0;
}
```

实数二分模板

```c++
#include<bits/stdc++.h>
#define ll long long
using namespace std;
ll f(int x){
    ll sum=0;
    for(...){
        if(...){
            sum+=...;
        }
    }
    return sum;
}
int main(){
    int l,r,ans,e=1e-5;
    while(l+e<=r){
        int mid=(l+r)/2;
        if(f(mid)>=x){
            l=mid;
        }
        else{
            r=mid;
        }
    }
    cout<<ans<<'\n';
    return 0;
}
```
