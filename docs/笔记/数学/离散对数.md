# 离散对数（BSGS）

## 问题

给定 `a, b, m`（$\\\\gcd  (a,m)=1$），求最小的 `x` 使 $a^x \equiv  b (\bmod  m)$。

## 大步小步法

令 $x = i\cdot t - j$（$t = \lceil \sqrt m\rceil $），预处理 $a^j$（小步），查 $b\cdot a^{i\cdot t}$（大步）。复杂度 `O(√m)`。

```c++
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
ll qpow(ll a, ll b, ll m){ ll r=1%m; a%=m; while(b){ if(b&1)r=r*a%m; a=a*a%m; b>>=1; } return r; }
ll bsgs(ll a, ll b, ll m) {
    a%=m; b%=m; ll t = sqrt(m)+1;
    unordered_map<ll,ll> tb; ll e = 1;
    for (ll j=0;j<t;++j){ if(!tb.count(e)) tb[e]=j; e = e*a%m; }
    ll at = qpow(a, t, m);
    ll now = b;
    for (ll i=0;i<=t;++i) {
        if (tb.count(now)) return i*t - tb[now];
        now = now*at%m;
    }
    return -1;
}
int main() { cout << bsgs(2, 8, 11) << '\n'; return 0; }  // 2^?=8 mod11 -> 3
```
