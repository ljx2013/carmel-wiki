# 中国剩余定理（CRT）

## 问题

求 `x` 满足模两两互质的同余方程组：

$$
x \equiv  a1 (\bmod  m1)
x \equiv  a2 (\bmod  m2)
…                      （\\\\gcd  (mi, mj)=1）
$$

## 构造解

令 $M = \prod  mi$，$Mi = M/mi$，$ti = Mi 在模 mi 下的逆元$，则

$$
x = \Sigma  ai \cdot  Mi \cdot  ti   (\bmod  M)
$$

```c++
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

ll exgcd(ll a, ll b, ll &x, ll &y){ if(!b){x=1;y=0;return a;} ll g=exgcd(b,a%b,y,x); y-=a/b*x; return g; }
ll inv(ll a, ll m){ ll x,y; exgcd(a,m,x,y); return (x%m+m)%m; }

// 返回解 x (mod M)；要求 mi 两两互质
ll crt(ll a[], ll m[], int n) {
    ll M = 1; for (int i=0;i<n;++i) M *= m[i];
    ll ans = 0;
    for (int i=0;i<n;++i) {
        ll Mi = M / m[i];
        ans = (ans + a[i] * Mi % M * inv(Mi, m[i])) % M;
    }
    return ans;
}

int main() {
    ll a[] = {2, 3, 2}, m[] = {3, 5, 7};
    cout << crt(a, m, 3) << '\n';   // 23
    return 0;
}
```
