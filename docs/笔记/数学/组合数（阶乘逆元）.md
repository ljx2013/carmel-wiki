# 组合数（阶乘逆元求法）

## 思想

当 `n` 较大、模 `MOD` 为质数时，用阶乘 + 逆元 $C(n,k) = n! \cdot  (k!)⁻¹ \cdot  ((n-k)!)⁻¹ \bmod  MOD$。

```c++
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
const ll MOD = 1000000007;
ll fac[200005], inv[200005];

ll qpow(ll a, ll b) { ll r=1%MOD; a%=MOD; while(b){ if(b&1) r=r*a%MOD; a=a*a%MOD; b>>=1; } return r; }

void init(int n) {
    fac[0] = 1;
    for (int i=1;i<=n;++i) fac[i] = fac[i-1] * i % MOD;
    inv[n] = qpow(fac[n], MOD-2);
    for (int i=n-1;i>=0;--i) inv[i] = inv[i+1] * (i+1) % MOD;
}

ll C(int n, int m) {
    if (m < 0 || m > n) return 0;
    return fac[n] * inv[m] % MOD * inv[n-m] % MOD;
}

int main() {
    init(100000);
    cout << C(100000, 50000) << '\n';
    return 0;
}
```
