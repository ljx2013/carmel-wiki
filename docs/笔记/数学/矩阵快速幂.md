# 矩阵快速幂

## 思想

把快速幂中的"数乘"换成"矩阵乘法"，用来在 `O(k^3 log n)` 内求第 `n` 项线性递推（如斐波那契）。

## 实现

```c++
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
const ll MOD = 1000000007;

struct Mat { ll a[2][2]; Mat(){ fill(a[0], a[0]+4, 0); } };

Mat mul(const Mat &x, const Mat &y) {
    Mat r;
    for (int i=0;i<2;++i) for (int j=0;j<2;++j)
        for (int k=0;k<2;++k) r.a[i][j] = (r.a[i][j] + x.a[i][k]*y.a[k][j]) % MOD;
    return r;
}

Mat mpow(Mat x, ll n) {
    Mat r; r.a[0][0]=r.a[1][1]=1;
    while (n) { if (n&1) r = mul(r, x); x = mul(x, x); n >>= 1; }
    return r;
}

int main() {
    Mat A; A.a[0][0]=A.a[0][1]=A.a[1][0]=1;   // 斐波那契转移矩阵
    Mat R = mpow(A, 10);
    cout << R.a[0][1] << '\n';              // F(11) = 89
    return 0;
}
```
