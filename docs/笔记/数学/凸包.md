# 凸包（Andrew 单调链）

## 思想

点按 x（再 y）排序，分别构造下凸壳与上凸壳，用叉积维护"向右转"。

```c++
#include <bits/stdc++.h>
using namespace std;
struct P { long long x, y; };
long long cross(P a, P b, P c){ return (b.x-a.x)*(c.y-a.y) - (b.y-a.y)*(c.x-a.x); }
bool cmp(P a, P b){ return a.x!=b.x ? a.x<b.x : a.y<b.y; }
int main() {
    vector<P> p = {{0,0},{2,0},{2,2},{0,2},{1,1}};
    sort(p.begin(), p.end(), cmp);
    vector<P> h;
    for (int k=0;k<2;++k) {
        int s = h.size();
        for (auto q : p) {
            while (h.size()>s && cross(h[h.size()-2], h.back(), q) <= 0) h.pop_back();
            h.push_back(q);
        }
        h.pop_back();
        reverse(p.begin(), p.end());
    }
    cout << h.size() << '\n';   // 凸包顶点数 4
    return 0;
}
```
