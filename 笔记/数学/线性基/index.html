    线性基¶

    思想¶

    用至多 60 个向量（按二进制位）表示给定数集的异或线性空间，支持：插入、查询最大/最小异或和、查询某数可否表出。

    实现（异或线性基）¶

    #include <bits/stdc++.h>
using namespace std;
using ll = long long;

struct Basis {
    ll b[60] = {};
    bool insert(ll x) {
        for (int i=59;i>=0;--i) if (x>>i&1) {
            if (b[i]) x ^= b[i];
            else { b[i] = x; return true; }
        }
        return false;   // x 已被表出
    }
    ll max_xor() { ll r=0; for (int i=59;i>=0;--i) if ((r^b[i])>r) r ^= b[i]; return r; }
};

int main() {
    Basis B; for (ll x : {1, 2, 4, 8}) B.insert(x);
    cout << B.max_xor() << '\n';   // 15
    return 0;
}

    小结¶

    操作 复杂度
    插入 / 查询最大 O(log V)
